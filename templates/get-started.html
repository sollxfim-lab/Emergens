# Oxysintx API — cURL Integration Example
# Base URL: http://node1.lunes.host:2381
# Authentication: Bearer token in Authorization header
# ─────────────────────────────────────────────

# Step 1: Start a scan — this returns a job_id immediately
curl -X POST http://node1.lunes.host:2381/api/scan/start \
  -H "Authorization: Bearer oxy_sk_live_69b215b08f66480295691864a0c9bf57bfed6903" \
  -H "Content-Type: application/json" \
  -d '{"target":"example.com","mode":"basic","tools":[]}'
# Response → { "job_id": "abc123" }

# Step 2: Poll for the result (repeat until status is "completed" or "failed")
curl http://node1.lunes.host:2381/api/scan/abc123/status \
  -H "Authorization: Bearer oxy_sk_live_69b215b08f66480295691864a0c9bf57bfed6903"
# Response → { "status": "completed", "results": { ... } }

# One-liner polling loop (bash):
while true; do
  RESP=$(curl -s http://node1.lunes.host:2381/api/scan/abc123/status \
    -H "Authorization: Bearer oxy_sk_live_69b215b08f66480295691864a0c9bf57bfed6903")
  STATUS=$(echo $RESP | jq -r '.status')
  if [ "$STATUS" = "completed" ] || [ "$STATUS" = "failed" ]; then
    echo $RESP | jq '.results'; break;
  fi;
  sleep 1;
done

// Oxysintx API — JavaScript (fetch) Integration
// Base URL: http://node1.lunes.host:2381
// Authentication: Bearer token in Authorization header
// ─────────────────────────────────────────────

const BASE = "http://node1.lunes.host:2381";
const TOKEN = "oxy_sk_live_69b215b08f66480295691864a0c9bf57bfed6903";

async function runScan(target) {
  // Step 1: Start the scan
  const start = await fetch(`${BASE}/api/scan/start`, {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${TOKEN}`,
      "Content-Type": "application/json"
    },
    body: JSON.stringify({ target, mode: "basic", tools: [] })
  }).then(r => r.json());

  // Step 2: Poll until completed or failed
  let job;
  do {
    await new Promise(r => setTimeout(r, 1000));
    job = await fetch(`${BASE}/api/scan/${start.job_id}/status`, {
      headers: { "Authorization": `Bearer ${TOKEN}` }
    }).then(r => r.json());
  } while (job.status !== "completed" && job.status !== "failed");

  // Step 3: Return the results
  return job.results;
}

// Usage:
runScan("example.com").then(results => console.log(results));
