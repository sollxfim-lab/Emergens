// Emergens API: JavaScript (fetch) Integration
// ─────────────────────────────────────────────

const BASE = "http://node1.lunes.host:2381";
const TOKEN = "emg_sk_live_...";

async function runScan(target) {
  const start = await fetch(`${BASE}/api/scan/start`, {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${TOKEN}`,
      "Content-Type": "application/json"
    },
    body: JSON.stringify({ target, mode: "basic" })
  }).then(r => r.json());

  let job;
  do {
    await new Promise(r => setTimeout(r, 1000));
    job = await fetch(`${BASE}/api/scan/${start.job_id}/status`, {
      headers: { "Authorization": `Bearer ${TOKEN}` }
    }).then(r => r.json());
  } while (job.status !== "completed");

  return job.results;
}

// Emergens API: PHP (cURL) Integration
// ─────────────────────────────────────────────

$BASE = "http://node1.lunes.host:2381";
$TOKEN = "emg_sk_live_...";

function apiRequest($method, $path, $data = null) {
  global $BASE, $TOKEN;
  $ch = curl_init($BASE . $path);
  $headers = ["Authorization: Bearer " . $TOKEN];
  curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
  curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
  if ($method === "POST") {
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
  }
  return json_decode(curl_exec($ch), true);
}

$start = apiRequest("POST", "/api/scan/start", ["target" => "example.com"]);
do {
  sleep(1);
  $job = apiRequest("GET", "/api/scan/" . $start["job_id"] . "/status");
} while (!in_array($job["status"], ["completed", "failed"]));

# Emergens API: Ruby (net/http) Integration
# ─────────────────────────────────────────────

require 'net/http'
require 'json'
require 'uri'

BASE = "http://node1.lunes.host:2381"
TOKEN = "emg_sk_live_..."

def api_request(method, path, data = nil)
  uri = URI(BASE + path)
  http = Net::HTTP.new(uri.host, uri.port)
  headers = {'Authorization' => "Bearer #{TOKEN}"}
  request = method == 'POST' ? Net::HTTP::Post.new(uri, headers) : Net::HTTP::Get.new(uri, headers)
  request.body = data.to_json if data
  JSON.parse(http.request(request).body)
end

start = api_request('POST', '/api/scan/start', {target: 'example.com'})
job = {}
loop do
  sleep 1
  job = api_request('GET', "/api/scan/#{start['job_id']}/status")
  break if ['completed', 'failed'].include?(job['status'])
end
